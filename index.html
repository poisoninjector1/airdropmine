async function syncData() { const statusElem = document.getElementById('api-status'); const statusText = document.getElementById('status-text'); // 1. Ensure user ID is always a valid Integer number let userId = user && user.id ? parseInt(user.id) : 99999; // 2. Parse referrerId safely let refId = null; if (referrerId) { let parsed = parseInt(referrerId.replace('ref_', '')); if (!isNaN(parsed)) refId = parsed; } try { const response = await fetch(`${BACKEND_URL}/api/user/sync`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ user_id: userId, username: user.username || "Guest", referrer_id: refId }) }); if (response.ok) { const data = await response.json(); currentBalance = data.balance; speedPerSec = data.mining_speed || 10; document.getElementById('speed').innerText = speedPerSec; statusElem.className = "status-container status-online"; statusText.innerText = "API Connected"; isApiConnected = true; } else { console.error("API response error:", await response.text()); throw new Error("API Server error"); } } catch (err) { console.error("Connection Error:", err); statusElem.className = "status-container status-offline"; statusText.innerText = "API Disconnected"; isApiConnected = false; } }
